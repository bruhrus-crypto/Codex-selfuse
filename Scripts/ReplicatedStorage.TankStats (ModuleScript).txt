local TankStats = {}

-- Вспомогательные функции
local function rpmToCooldown(rpm)
	if type(rpm) ~= "number" or rpm <= 0 then return math.huge end
	return 60 / rpm
end

local function calcEffectiveDamage(baseDamage, armor)
	if type(armor) ~= "number" or armor <= 0 then return baseDamage end
	return baseDamage * (100 / (100 + armor))
end

local function calcArmorPenetration(damage, armor)
	if type(armor) ~= "number" or armor <= 0 then return true end
	if type(damage) ~= "number" or damage <= 0 then return false end
	local penetrationChance = math.min(damage / (armor * 2), 0.95)
	return math.random() < penetrationChance
end
